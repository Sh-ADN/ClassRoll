# Project-wide Gradle settings.
# Memory for the Gradle daemon. Kotlin compiles in-process (see below), so it needs
# enough heap for Kotlin + KSP + R8 together; 2 GB is too tight for this project.
org.gradle.jvmargs=-Xmx4g -Dfile.encoding=UTF-8
android.useAndroidX=true
kotlin.code.style=official
android.nonTransitiveRClass=true
# Avoid the "Could not connect to Kotlin compile daemon" error on CI machines.
kotlin.compiler.execution.strategy=in-process
# Let the google-services plugin run even if google-services.json is not present.
googleServices.missing.passthrough=true
