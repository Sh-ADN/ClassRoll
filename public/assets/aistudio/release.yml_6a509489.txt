name: Build & Release APK

# Runs every time AI Studio (or you) pushes to main.
# Also runnable manually from the Actions tab.
on:
  push:
    branches: [ main ]
  workflow_dispatch: {}

# Needed so the workflow is allowed to create a Release.
permissions:
  contents: write

jobs:
  build-and-release:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Set up JDK 21
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'

      - name: Setup Gradle
        uses: gradle/actions/setup-gradle@v3

      - name: Accept Android SDK Licenses
        run: |
          mkdir -p "$ANDROID_HOME/licenses"
          echo -e "24333f8a637b3a4e1706d3132e0bf49ab4633632\n8933902d0775e460b80f4e5a0e283a7d11413d29\nd56f51508286000c7c158107f7948d14e5628597\n60100772780c44369179421288e0c07a1469782f\nf39934297f000e3a3273702c23410b63b156901d\n84831b9409646a918e30573bab4c9c91346d8abd\n504667f4c05772d9a158b1713028d7845763a725\n8e5425713d0c1de0b40e3262f741544aa8495a1a" > "$ANDROID_HOME/licenses/android-sdk-license"
          echo -e "84831b9409646a918e30573bab4c9c91346d8abd" > "$ANDROID_HOME/licenses/android-sdk-preview-license"
          echo -e "33b622f7cb01413d70a3e6f6a0f45a7424f70401" > "$ANDROID_HOME/licenses/android-googletv-license"
          echo -e "60100772780c44369179421288e0c07a1469782f" > "$ANDROID_HOME/licenses/android-sdk-arm-dbt-license"
          echo -e "854734e56d36da5c68057d13e77d9a0b0825305f" > "$ANDROID_HOME/licenses/google-gdk-license"
          echo -e "e96678292d8ac641b2d23e3ea62341f012b70f00" > "$ANDROID_HOME/licenses/mips-android-sysimage-license"
          if [ -d "$ANDROID_HOME/cmdline-tools/latest/bin" ]; then
            yes | $ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager --licenses || true
          fi

      - name: Grant execute permission for gradlew
        run: chmod +x gradlew

      - name: Setup Keystore for Signing
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
          STORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
        run: |
          if [ -n "$KEYSTORE_BASE64" ]; then
            echo "Decoding repository secret KEYSTORE_BASE64 to my-upload-key.jks..."
            echo "$KEYSTORE_BASE64" | base64 -d > my-upload-key.jks
            # build.gradle.kts always looks for the alias "upload". If your stored
            # key was created with a different alias, rename it inside this
            # temporary copy. The key itself (your signing identity) is unchanged.
            ALIAS=$(keytool -list -keystore my-upload-key.jks -storepass "$STORE_PASSWORD" 2>/dev/null | awk -F, '/PrivateKeyEntry/ {print $1; exit}')
            if [ -n "$ALIAS" ] && [ "$ALIAS" != "upload" ]; then
              echo "Renaming key alias '$ALIAS' to 'upload' in the temporary build copy..."
              keytool -changealias -keystore my-upload-key.jks -storepass "$STORE_PASSWORD" -alias "$ALIAS" -destalias upload
            fi
          elif [ -f "debug.keystore.base64" ]; then
            echo "Decoding debug.keystore.base64 to debug.keystore..."
            base64 -d debug.keystore.base64 > debug.keystore
          else
            echo "::warning::No KEYSTORE_BASE64 secret set. Generating a throwaway key, so each build is signed differently and cannot be installed over the previous one."
            keytool -genkeypair -v -keystore debug.keystore -alias androiddebugkey \
              -keyalg RSA -keysize 2048 -validity 10000 \
              -storepass android -keypass android \
              -dname "CN=ClassRoll, OU=App, O=AIStudio, L=City, ST=State, C=US"
          fi

          echo "=== RELEASE SIGNING KEY FINGERPRINTS ==="
          if [ -f "my-upload-key.jks" ]; then
            keytool -list -v -keystore my-upload-key.jks -storepass "$STORE_PASSWORD" 2>/dev/null | grep -E "SHA1:|SHA256:" || true
          elif [ -f "debug.keystore" ]; then
            keytool -list -v -keystore debug.keystore -storepass android 2>/dev/null | grep -E "SHA1:|SHA256:" || true
          fi

      # settings.gradle.kts and app/build.gradle.kts already create a default
      # google-services.json automatically when none exists, so no inline JSON
      # is needed here. This step only restores a real one if you provide it
      # as the GOOGLE_SERVICES_JSON repository secret.
      - name: Setup Google Services Configuration
        env:
          GOOGLE_SERVICES_JSON: ${{ secrets.GOOGLE_SERVICES_JSON }}
        run: |
          if [ -n "$GOOGLE_SERVICES_JSON" ]; then
            echo "Restoring google-services.json from repository secret..."
            printf '%s' "$GOOGLE_SERVICES_JSON" > app/google-services.json
          else
            echo "No GOOGLE_SERVICES_JSON secret set; the build will use the default generated by Gradle."
          fi

      # AI Studio regenerates the Gradle files each time you sync, and the regenerated
      # files have repeatedly disagreed with each other (an entry missing from the
      # version catalog, a dependency switched off that the code still needs). Either
      # one makes the build fail with a long, confusing Gradle error. This step checks
      # for those mismatches up front, repairs the two known ones for this build only
      # (your repository files are not changed), and otherwise fails with a clear message.
      - name: Check build files (auto-fix known export problems)
        run: |
          python3 - <<'PY'
          import pathlib, re, sys, tomllib

          # Icons that ship in the small "core" icon library (from Google's API listing).
          CORE_ICONS = set("""AccountBox AccountCircle AddCircle Add ArrowBack ArrowDropDown
          ArrowForward Build Call CheckCircle Check Clear Close Create DateRange Delete Done Edit
          Email ExitToApp Face FavoriteBorder Favorite Home Info KeyboardArrowDown KeyboardArrowLeft
          KeyboardArrowRight KeyboardArrowUp List LocationOn Lock MailOutline Menu MoreVert
          Notifications Person Phone Place PlayArrow Refresh Search Send Settings Share ShoppingCart
          Star ThumbUp Warning""".split())

          app_path = pathlib.Path("app/build.gradle.kts")
          cat_path = pathlib.Path("gradle/libs.versions.toml")

          def live_lines(text):
              return [l for l in text.splitlines() if not l.strip().startswith("//")]

          def unresolved(text):
              cat = tomllib.loads(cat_path.read_text())
              libs = {k.replace("-", ".") for k in cat.get("libraries", {})}
              plugins = {k.replace("-", ".") for k in cat.get("plugins", {})}
              live = "\n".join(live_lines(text))
              bad = [m.group(0) for m in re.finditer(r"libs\.plugins\.([\w.]+)", live) if m.group(1) not in plugins]
              bad += ["libs." + m.group(1) for m in re.finditer(r"libs\.(?!plugins\.|versions\.)([\w.]+)", live) if m.group(1) not in libs]
              return bad

          app_text = app_path.read_text()
          changed = False
          problems = []

          # 1. Entries used in app/build.gradle.kts that the version catalog does not define.
          bad = unresolved(app_text)
          roborazzi = [b for b in bad if "roborazzi" in b]
          other = [b for b in bad if "roborazzi" not in b]
          if roborazzi:
              print("::warning::app/build.gradle.kts uses Roborazzi (screenshot-test tooling) but gradle/libs.versions.toml no longer defines it. Skipping those lines for this build.")
              kept = []
              for line in app_text.splitlines():
                  is_comment = line.strip().startswith("//")
                  if not is_comment and ("libs.plugins.roborazzi" in line or "libs.roborazzi" in line):
                      continue
                  kept.append(line)
              app_text = "\n".join(kept) + "\n"
              changed = True
          if other:
              problems.append("app/build.gradle.kts uses these version-catalog entries, but gradle/libs.versions.toml does not define them: " + ", ".join(other))

          # 2. Icons that are not in the core icon library need the extended library.
          used = set()
          for f in pathlib.Path("app/src/main").rglob("*.kt"):
              t = f.read_text()
              used |= set(re.findall(r"import androidx\.compose\.material\.icons\.(?:automirrored\.)?(?:filled|outlined|rounded|sharp|twotone)\.(\w+)", t))
              used |= set(re.findall(r"Icons\.(?:AutoMirrored\.)?(?:Default|Filled|Outlined|Rounded|Sharp|TwoTone)\.(\w+)", t))
          extended_only = sorted(i for i in used if i not in CORE_ICONS)
          ext_active = any(re.match(r"\s*implementation\(libs\.androidx\.compose\.material\.icons\.extended\)", l) for l in live_lines(app_text))
          if extended_only and not ext_active:
              print("::warning::The app uses icons that only exist in the extended icon library (" + ", ".join(extended_only) + ") but that dependency is switched off. Switching it on for this build.")
              new_text = re.sub(r"^(\s*)//\s*(implementation\(libs\.androidx\.compose\.material\.icons\.extended\))", r"\1\2", app_text, flags=re.M)
              if new_text == app_text:
                  problems.append("The app uses extended-only icons but the material-icons-extended dependency line was not found in app/build.gradle.kts.")
              else:
                  app_text = new_text
                  changed = True

          if changed:
              app_path.write_text(app_text)
          if problems:
              for p in problems:
                  print("::error::" + p)
              sys.exit(1)
          print("Build file check passed." + (" (known problems were auto-fixed for this build; see the warnings above)" if changed else ""))
          PY

      - name: Build Release APK
        env:
          STORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: |
          # Memory and compiler settings are passed here (not only in gradle.properties)
          # because the regenerated gradle.properties dropped them, and the build then
          # runs short of memory on the GitHub runner.
          ./gradlew assembleRelease --stacktrace --no-daemon \
            -Dorg.gradle.jvmargs="-Xmx4g -Dfile.encoding=UTF-8" \
            -Pkotlin.compiler.execution.strategy=in-process

      - name: Locate & Prepare Release APK
        id: prepare_apk
        run: |
          mkdir -p release_out
          APK_FILE=$(find app/build/outputs/apk/release -name "*.apk" | head -n 1)
          if [ -z "$APK_FILE" ]; then
            echo "Error: No release APK found in app/build/outputs/apk/release!"
            exit 1
          fi
          echo "Found release APK: $APK_FILE"
          FINAL_APK="release_out/ClassRoll-build-${{ github.run_number }}.apk"
          cp "$APK_FILE" "$FINAL_APK"
          echo "apk_path=$FINAL_APK" >> "$GITHUB_OUTPUT"

      - name: Upload APK Artifact
        uses: actions/upload-artifact@v4
        with:
          name: ClassRoll-APK
          path: ${{ steps.prepare_apk.outputs.apk_path }}
          retention-days: 30

      - name: Set Release Tag
        id: vars
        run: echo "tag=build-${{ github.run_number }}" >> "$GITHUB_OUTPUT"

      - name: Publish GitHub Release
        uses: softprops/action-gh-release@v2
        if: startsWith(github.ref, 'refs/heads/main') || github.event_name == 'workflow_dispatch'
        with:
          tag_name: ${{ steps.vars.outputs.tag }}
          name: "ClassRoll – build ${{ github.run_number }}"
          make_latest: true
          files: ${{ steps.prepare_apk.outputs.apk_path }}
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
